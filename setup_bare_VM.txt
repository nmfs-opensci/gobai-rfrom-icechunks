# Setup for a bare VM, then process the netCDFs and upload them to NODD
#
# Paste these lines into a terminal a block at a time. Don't run the file with
# bash: `exec -l $SHELL` restarts the shell partway through, and
# `gcloud auth application-default login` waits for you to finish in a browser.
# setup.md explains each step in more detail.

# Needed on every VM, including one cloned from an image (gh is not carried over)
sudo apt-get update && sudo apt-get install -y gh

#### If you cloned the VM from one already set up, skip down to "Update the repo".

sudo apt-get install -y git curl tmux python3-venv python3-pip
python3 --version                                  # must be 3.11+

git clone https://github.com/nmfs-opensci/gobai-rfrom-icechunks.git
cd ~/gobai-rfrom-icechunks                         # nodd.py and requirements*.txt are at the repo root

git config --global user.email "your@email"
git config --global user.name "Your Name"
git config --global pull.rebase false

python3 -m venv .venv
source .venv/bin/activate
pip install -U pip
pip install -r requirements.txt                    # enough for nodd.py
pip install -r requirements-icechunk.txt           # only if this VM will also build the Icechunk store

# install gcloud
curl -fsSL https://sdk.cloud.google.com | bash     # then restart the shell:
exec -l $SHELL

# Opens a browser and saves the token to ~/.config/gcloud/application_default_credentials.json.
# On a VM with no browser, add --no-browser and follow the instructions it prints.
gcloud auth application-default login

#### End of first-time setup.

# Update the repo (every time, including on a cloned VM)
cd ~/gobai-rfrom-icechunks
git pull
source .venv/bin/activate

# Settings for every new shell and every tmux pane
export NODD_SCRATCH_DIR="$HOME/nodd-scratch"      # needs ~35 GB free; use an attached data disk if home is small
export NODD_GCS_TOKEN="$HOME/.config/gcloud/application_default_credentials.json"
df -h "$HOME"                                      # check the free space

# Check write access before a long run. Swap in noaa-oar-gobai / netcdf/v202606 for GOBAI.
python -c "
import gcsfs, os
fs = gcsfs.GCSFileSystem(token=os.environ['NODD_GCS_TOKEN'])
print(fs.ls('noaa-oar-rfrom/netcdf/v2.3/temp')[:3])   # read
fs.pipe('noaa-oar-rfrom/netcdf/v2.3/_write_check.txt', b'ok'); print('write OK')
fs.rm('noaa-oar-rfrom/netcdf/v2.3/_write_check.txt')
"

# Check the plan: prints the blocks, downloads nothing
python nodd.py --stream temp --list

# Run it inside tmux so a dropped SSH session doesn't kill it
tmux new -s nodd          # detach: Ctrl-b d    reattach: tmux attach -t nodd
cd ~/gobai-rfrom-icechunks && source .venv/bin/activate
export NODD_SCRATCH_DIR="$HOME/nodd-scratch"
export NODD_GCS_TOKEN="$HOME/.config/gcloud/application_default_credentials.json"
